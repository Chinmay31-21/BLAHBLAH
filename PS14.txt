PS14 — Display MICROCONTROLLER and LAB ESE
#include <xc.h>

#define RS RE0
#define EN RE1

void delay()
{
    unsigned int i;
    for(i=0; i<1000; i++);
}

void cmd(unsigned char a)
{
    RS = 0;
    PORTD = (PORTD & 0x0F) | (a & 0xF0);
    EN = 1; delay(); EN = 0;

    PORTD = (PORTD & 0x0F) | (a << 4);
    EN = 1; delay(); EN = 0;
}

void data(unsigned char a)
{
    RS = 1;
    PORTD = (PORTD & 0x0F) | (a & 0xF0);
    EN = 1; delay(); EN = 0;

    PORTD = (PORTD & 0x0F) | (a << 4);
    EN = 1; delay(); EN = 0;
}

void main()
{
    char s1[] = "MICROCONTROLLER";
    char s2[] = "LAB ESE";
    int i;

    ADCON1 = 0x0F;
    TRISD = 0x00;
    TRISE = 0x00;

    cmd(0x02);
    cmd(0x28);
    cmd(0x0C);
    cmd(0x06);
    cmd(0x80);

    for(i=0; s1[i]!='\0'; i++)
        data(s1[i]);

    cmd(0xC0);

    for(i=0; s2[i]!='\0'; i++)
        data(s2[i]);

    while(1);
}
