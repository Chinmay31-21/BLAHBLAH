Using the LPC2148 microcontroller, configure the on-chip ADC to read an analog input
and transmit both the ADC count and the corresponding voltage value to the serial
terminal.

#CODE


#include <LPC214X.H>

void uart_init(void)
{
    PINSEL0 = 0x00000005;
    U0LCR = 0x83;
    U0DLM = 0x00;
    U0DLL = 0x61;
    U0LCR = 0x03;
}

void uart_char(char ch)
{
    while(!(U0LSR & 0x20));
    U0THR = ch;
}

void uart_string(char *str)
{
    while(*str)
        uart_char(*str++);
}

void uart_number(unsigned int num)
{
    char digits[10];
    int i = 0;

    if(num == 0)
    {
        uart_char('0');
        return;
    }

    while(num > 0)
    {
        digits[i++] = (num % 10) + '0';
        num = num / 10;
    }

    while(i > 0)
        uart_char(digits[--i]);
}

int main(void)
{
    unsigned int adc, mv;
    volatile unsigned int i, j;

    PINSEL1 &= ~(3U << 24);
    PINSEL1 |= (1U << 24);

    uart_init();

    AD0CR = (1U << 1) | (3U << 8) | (1U << 21);

    while(1)
    {
        AD0CR &= ~(7U << 24);
        AD0CR |= (1U << 24);

        while((AD0GDR & (1U << 31)) == 0);

        adc = (AD0GDR >> 6) & 0x3FF;

        mv = (adc * 3300UL) / 1023;

        uart_string("ADC Count: ");
        uart_number(adc);

        uart_string("  Voltage: ");
        uart_number(mv / 1000);
        uart_char('.');
        
        if((mv % 1000) < 100)
            uart_char('0');
        if((mv % 1000) < 10)
            uart_char('0');

        uart_number(mv % 1000);
        uart_string(" V\r\n");

        for(i = 0; i < 500; i++)
            for(j = 0; j < 1000; j++);
    }
}
