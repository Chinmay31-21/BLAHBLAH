PS5 — Continuously transmit name and roll number
#include <reg51.h>

void main()
{
    char s[] = "Chinmay, Roll No: 123\n";
    unsigned int i;

    TMOD = 0x20;
    TH1 = 0xFD;
    SCON = 0x50;
    TR1 = 1;

    while(1)
    {
        for(i=0; s[i]!='\0'; i++)
        {
            SBUF = s[i];
            while(TI == 0);
            TI = 0;
        }
    }
}
