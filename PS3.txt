PS3 — Two cycles of 400 Hz followed by two cycles of 900 Hz
#include <reg51.h>

sbit OUT = P1^0;

void main()
{
    unsigned int i;

    TMOD = 0x10;

    while(1)
    {
        for(i=0; i<4; i++)
        {
            OUT = ~OUT;
            TH1 = 0xFB;
            TL1 = 0x1E;
            TF1 = 0;
            TR1 = 1;
            while(TF1 == 0);
            TR1 = 0;
        }

        for(i=0; i<4; i++)
        {
            OUT = ~OUT;
            TH1 = 0xFD;
            TL1 = 0xF0;
            TF1 = 0;
            TR1 = 1;
            while(TF1 == 0);
            TR1 = 0;
        }
    }
}
