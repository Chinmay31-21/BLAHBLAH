Using the LPC2148 microcontroller, configure the on-chip RTC to display the current time on
a 16x2 LCD and turn ON an LED whenever the seconds value reaches 30

#include <LPC214X.H>
#include <stdio.h>

#define RS       (1U << 11)
#define EN       (1U << 10)
#define RW       (1U << 20)
#define LCD_DATA 0x00FF0000
#define LED      (1U << 31)

char data[20];

void delay(unsigned int count)
{
    unsigned int i, j;

    for(i = 0; i < count; i++)
        for(j = 0; j < 10; j++);
}

void lcd_cmd(unsigned char cmd)
{
    IOCLR0 = RS | RW;
    IOCLR1 = LCD_DATA;
    IOSET1 = ((unsigned int)cmd << 16);

    IOSET0 = EN;
    delay(10);
    IOCLR0 = EN;
    delay(100);
}

void lcd_dat(unsigned char dat)
{
    IOSET0 = RS;
    IOCLR0 = RW;
    IOCLR1 = LCD_DATA;
    IOSET1 = ((unsigned int)dat << 16);

    IOSET0 = EN;
    delay(10);
    IOCLR0 = EN;
    delay(100);
}

void lcd_str(char *str)
{
    while(*str)
        lcd_dat(*str++);
}

void lcd_init(void)
{
    PINSEL0 &= ~((3U << 20) | (3U << 22));
    PINSEL1 &= ~(3U << 8);
    PINSEL2 &= ~(1U << 3);

    IODIR0 |= RS | EN | RW;
    IODIR1 |= LCD_DATA | LED;

    IOCLR0 = RS | EN | RW;
    IOCLR1 = LCD_DATA | LED;

    delay(2000);

    lcd_cmd(0x38);
    lcd_cmd(0x0C);
    lcd_cmd(0x06);
    lcd_cmd(0x01);

    delay(200);
}

void rtc_init(void)
{
    CCR = 0x02;
    CCR = 0x00;

    PREINT = 456;
    PREFRAC = 25024;

    SEC = 0;
    MIN = 9;
    HOUR = 10;

    CCR = 0x01;
}

int main(void)
{
    unsigned int h, m, s;

    lcd_init();
    rtc_init();

    while(1)
    {
        h = HOUR;
        m = MIN;
        s = SEC;

        sprintf(data, "TIME: %02d:%02d:%02d",
                (int)h, (int)m, (int)s);

        lcd_cmd(0x80);
        lcd_str(data);

        lcd_cmd(0xC0);
        lcd_str("LPC2148 RTC     ");

        if(s == 30)
        {
            IOSET1 = LED;
        }
        else
        {
            IOCLR1 = LED;
        }
    }
}
