PS18–PS21: PIC18F4550 keypad and LCD
Keypad connections: 4×4 keypad on PORTB (RB0–RB3 rows, RB4–RB7 columns). LCD uses the PS13 connections (RD4–RD7, RE0 and RE1).

Use this common keypad and LCD code for PS18–PS21. The getkey() function returns the pressed key as an ASCII character.
#include <xc.h>

#define RS RE0
#define EN RE1

void delay()
{
    unsigned int i;
    for(i=0; i<1000; i++);
}

void cmd(unsigned char a)
{
    RS = 0;
    PORTD = (PORTD & 0x0F) | (a & 0xF0);
    EN = 1; delay(); EN = 0;
    PORTD = (PORTD & 0x0F) | (a << 4);
    EN = 1; delay(); EN = 0;
}

void data(unsigned char a)
{
    RS = 1;
    PORTD = (PORTD & 0x0F) | (a & 0xF0);
    EN = 1; delay(); EN = 0;
    PORTD = (PORTD & 0x0F) | (a << 4);
    EN = 1; delay(); EN = 0;
}

void lcd_init()
{
    cmd(0x02);
    cmd(0x28);
    cmd(0x0C);
    cmd(0x06);
}

char getkey()
{
    char keys[4][4] = {
        {'1','2','3','A'},
        {'4','5','6','B'},
        {'7','8','9','C'},
        {'*','0','#','D'}
    };

    unsigned char row, col;

    TRISB = 0xF0;

    while(1)
    {
        for(row=0; row<4; row++)
        {
            PORTB = ~(1 << row);
            delay();

            for(col=0; col<4; col++)
            {
                if((PORTB & (0x10 << col)) == 0)
                {
                    while((PORTB & (0x10 << col)) == 0);
                    return keys[row][col];
                }
            }
        }
    }
}


PS18 — Detect one key and display its character
void main()
{
    char key;

    ADCON1 = 0x0F;
    TRISD = 0x00;
    TRISE = 0x00;

    lcd_init();

    while(1)
    {
        key = getkey();

        cmd(0x80);
        data(key);
    }
}


PS19 — Read four successive key presses
void main()
{
    char key;
    unsigned char i;

    ADCON1 = 0x0F;
    TRISD = 0x00;
    TRISE = 0x00;

    lcd_init();
    cmd(0x80);

    for(i=0; i<4; i++)
    {
        key = getkey();
        data(key);
    }

    while(1);
}


PS20 — Display pressed digit and its ASCII code
void main()
{
    char key;
    unsigned char ascii;

    ADCON1 = 0x0F;
    TRISD = 0x00;
    TRISE = 0x00;

    lcd_init();

    while(1)
    {
        key = getkey();

        if(key >= '0' && key <= '9')
        {
            ascii = key;

            cmd(0x80);
            data('D'); data(':'); data(key);

            cmd(0xC0);
            data('A'); data(':');
            data((ascii/100) + '0');
            data(((ascii/10)%10) + '0');
            data((ascii%10) + '0');
        }
    }
}


PS21 — Display VALID or FUNCTION
void main()
{
    char key;
    char s1[] = "VALID";
    char s2[] = "FUNCTION";
    unsigned char i;

    ADCON1 = 0x0F;
    TRISD = 0x00;
    TRISE = 0x00;

    lcd_init();

    while(1)
    {
        key = getkey();
        cmd(0x01);
        cmd(0x80);

        if(key >= '0' && key <= '9')
        {
            for(i=0; s1[i]!='\0'; i++)
                data(s1[i]);
        }
        else if(key >= 'A' && key <= 'D')
        {
            for(i=0; s2[i]!='\0'; i++)
                data(s2[i]);
        }
    }
}



PS22–PS25: PIC18F4550 stepper motor programs
Assumptions: Stepper motor is connected through a suitable driver (ULN2003 or equivalent), with driver inputs on RD0–RD3. The sequence below is a four-step wave-drive sequence. The number of steps per revolution depends on your motor and its gearing.
PS22 — Continuous clockwise rotation
#include <xc.h>

void delay()
{
    unsigned int i;
    for(i=0; i<50000; i++);
}

void main()
{
    unsigned char a[4] = {0x01,0x02,0x04,0x08};
    unsigned char i;

    ADCON1 = 0x0F;
    TRISD = 0xF0;

    while(1)
    {
        for(i=0; i<4; i++)
        {
            PORTD = a[i];
            delay();
        }
    }
}

PS24 — Clockwise or anticlockwise using two buttons
Connections: Clockwise button on RA0, anticlockwise button on RA1.
#include <xc.h>

void delay()
{
    unsigned int i;
    for(i=0; i<50000; i++);
}

void main()
{
    unsigned char a[4] = {0x01,0x02,0x04,0x08};
    unsigned char i;

    ADCON1 = 0x0F;
    TRISD = 0xF0;
    TRISA = 0x03;

    while(1)
    {
        if(RA0 == 0)
        {
            for(i=0; i<4; i++)
            {
                PORTD = a[i];
                delay();
            }
        }

        if(RA1 == 0)
        {
            for(i=4; i>0; i--)
            {
                PORTD = a[i-1];
                delay();
            }
        }
    }
}


PS25 — One revolution clockwise, pause, then anticlockwise
Example: 200 steps per revolution. Change 200 according to your motor's actual steps per revolution.
#include <xc.h>

void delay()
{
    unsigned int i;
    for(i=0; i<50000; i++);
}

void main()
{
    unsigned char a[4] = {0x01,0x02,0x04,0x08};
    unsigned int i;

    ADCON1 = 0x0F;
    TRISD = 0xF0;

    while(1)
    {
        for(i=0; i<200; i++)
        {
            PORTD = a[i%4];
            delay();
        }

        PORTD = 0x00;
        delay();

        for(i=0; i<200; i++)
        {
            PORTD = a[3-(i%4)];
            delay();
        }

        PORTD = 0x00;
        delay();
    }
}
